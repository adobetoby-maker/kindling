## Method (not run in this session)

Save the script as `editor/MOVEMENT-005-ENSEMBLE-METRICS.py` and run it from `books/book-02-meridian`.

Notes on the definitions:
- The lexicons are deliberately narrow and are published in the script.
- Words like "draw" and "set" have ordinary senses, so treat hits as proxies.
- An -ly hit is not proof that the word is an adverb.

```python
#!/usr/bin/env python3
"""Movement 5 ensemble metrics. Run from books/book-02-meridian."""
import re, statistics as st
from collections import Counter
from pathlib import Path

FILES = [Path(f"revisions/ensemble-2026-09-29/manuscript/chapter-{n}.md") for n in range(33, 41)]
LEX = {  # published definitions; proxies, not proof
    "progression": r"\b(?:handfuls?|flasks?|settl(?:e|ed|es|ing)|(?:my|your|his|her|their) (?:floor|stone)|struts?|vent(?:s|ed|ing)?|draw(?:s|n|ing)?|drew|lee)\b",
    "ly_suffix": r"\b[a-z]+ly\b",
    "reporting": r"\b(?:said|says|asked|called|answered|shouted|whispered|told)\b",
    "combat": r"\b(?:kill(?:s|ed|ing)?|strike|struck|spike|husks?|fight|hit)\b",
    "that": r"\bthat\b", "just": r"\bjust\b", "almost": r"\balmost\b",
    "felt": r"\bfelt\b", "seemed": r"\bseemed\b",
}
WORD = re.compile(r"[A-Za-z0-9]+(?:['’][A-Za-z]+)?")

def syllables(w):  # vowel-group estimator, silent-e adjusted
    w = w.lower()
    g = re.findall(r"[aeiouy]+", w)
    n = len(g) - (1 if len(g) > 1 and w.endswith("e") and not w.endswith(("le", "ee")) else 0)
    return max(1, n)

text = "\n".join("\n".join(l for l in p.read_text().splitlines() if not l.startswith("#")) for p in FILES)
blocks = [b.strip() for b in re.split(r"\n\s*\n", text) if b.strip()]
breaks = sum(b == "---" for b in blocks)
paras = [b for b in blocks if b != "---"]
flat = re.sub(r"\s+", " ", " ".join(paras))
sents = [s for s in re.split(r"(?<=[.!?])[\"”’*_)]*\s+", flat) if WORD.search(s)]
slen = [len(WORD.findall(s)) for s in sents]
words = WORD.findall(flat); N, S = len(words), len(sents)
syl = sum(syllables(w) for w in words)
plen = [len(WORD.findall(p)) for p in paras]
print(f"words {N} sentences {S} paragraphs {len(paras)} breaks {breaks} ({breaks/N*1e4:.1f}/10k)")
print(f"sent mean {st.mean(slen):.1f} median {st.median(slen)} pstdev {st.pstdev(slen):.1f} "
      f"<=5 {sum(l <= 5 for l in slen)/S:.1%} >=40 {sum(l >= 40 for l in slen)/S:.1%}")
print(f"para mean {st.mean(plen):.1f} median {st.median(plen)}")
print(f"Flesch {206.835 - 1.015*N/S - 84.6*syl/N:.1f}  FK {0.39*N/S + 11.8*syl/N - 15.59:.1f}")
low = flat.lower()
for k, rx in LEX.items():
    print(f"{k:12} {len(re.findall(rx, low))/N*1e4:.1f} per 10k")
toks = WORD.findall(low)
print("top 3-grams:", [" ".join(t) for t, _ in Counter(zip(toks, toks[1:], toks[2:])).most_common(10)])
```
